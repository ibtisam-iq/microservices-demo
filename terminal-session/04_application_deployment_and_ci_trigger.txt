# ArgoCD Helm Values Patch for EKS + Gateway API

# 1. Run ArgoCD server in insecure (HTTP) mode
# TLS terminates at the ALB (via ACM certificate). The ALB forwards plain
# HTTP to ArgoCD internally.
configs:
  params:
    server.insecure: true

# 2. Enable Helm rendering inside Kustomize
# ArgoCD Applications use kustomization.yaml files that reference helmCharts: blocks.
# Remove this line if all ArgoCD Applications use pure Helm sources.
  cm:
    kustomize.buildOptions: "--enable-helm"

# 3. Expose ArgoCD server via Gateway API HTTPRoute
# The Helm chart natively supports creating an HTTPRoute resource. This
# attaches to your existing Gateway and tells the ALB to route traffic
# for argocd.ibtisam.qzz.io to the argocd-server Service.
server:
  httproute:
    enabled: true
    parentRefs:
      - name: app-alb-gateway
        namespace: default
        sectionName: https
    hostnames:
      - argocd.ibtisam.qzz.io
    rules:
      - matches:
          - path:
              type: PathPrefix
              value: /

# ---------------------------------------------------------------
# HTTPRoute: Online Boutique Frontend
#
# Routes external traffic to the frontend Service created by the
# onlineboutique Helm chart. The chart's frontend.externalService
# is set to false in values-eks.yaml, so no LoadBalancer Service
# exists. This HTTPRoute + the ALB Gateway is the sole ingress path.
#
# parentRefs bind to the shared ALB Gateway (app-alb-gateway) in
# the default namespace. Both HTTP and HTTPS listeners are attached
# so the ALB handles TLS termination via ACM.
#
# backendRef targets the "frontend" Service on port 80, which is
# the ClusterIP Service the Helm chart creates when
# frontend.externalService: false.
#
# group, kind, weight, and matches are Gateway API defaults that
# the ALB controller injects after creation. Including them here
# explicitly prevents ArgoCD from showing perpetual OutOfSync drift
# between the desired state (Git) and the live state (cluster).
# ---------------------------------------------------------------
apiVersion: gateway.networking.k8s.io/v1beta1
kind: HTTPRoute
metadata:
  name: http-app-route
  namespace: boutique-app
spec:
  hostnames:
    - "app.ibtisam.qzz.io"
  parentRefs:
  - group: gateway.networking.k8s.io
    namespace: default
    kind: Gateway
    name: app-alb-gateway
    sectionName: http
  - group: gateway.networking.k8s.io
    namespace: default
    kind: Gateway
    name: app-alb-gateway
    sectionName: https
  rules:
  - backendRefs:
    - group: ""
      kind: Service
      name: frontend
      port: 80
      weight: 1
    matches:
    - path:
        type: PathPrefix
        value: /

# TargetGroupConfiguration: Frontend Pod IP Targeting
#
# Configures the AWS ALB to route directly to frontend pod IPs
# (targetType: ip) instead of NodePort. This works because EKS
# nodes and pods share the VPC CIDR, so the ALB can reach pod IPs
# natively without an extra hop through kube-proxy.
#
# targetReference.name matches the "frontend" Service created by
# the onlineboutique Helm chart. The ALB Load Balancer Controller
# uses this binding to register frontend pod IPs as targets in the
# ALB target group.
# ---------------------------------------------------------------

kind: TargetGroupConfiguration

  name: app-tg-config
  namespace: boutique-app
spec:
  targetReference:
    name: frontend
  defaultConfiguration:
    targetType: ip

# ---------------------------------------------------------------
# ImageUpdater: microservices-demo
#
# Watches the microservices-demo ArgoCD Application and tracks
# all 10 service images on GHCR. Uses the digest strategy to
# detect when CI pushes a new image behind the :latest tag.
#
# Update strategy: digest
#   Compares the digest of the :latest tag in GHCR against what
#   is currently running in the cluster. When CI pushes a new
#   build, the digest behind :latest changes, Image Updater
#   detects it and patches the ArgoCD Application with the new
#   digest. No image creation timestamps involved.
#
#   newest-build was evaluated but does not work reliably with
#   BuildKit-produced images because BuildKit sets the image
#   config's "created" field to epoch (1970-01-01T00:00:00Z)
#   for reproducible builds, making all tags appear identical
#   in age.
#
# Update method: argocd (default)
#   Image overrides are patched directly onto the Application CR
#   as kustomize image entries. No Git write-back. The override
#   persists in the Application object. If the Application is
#   ever deleted and recreated from Git, Image Updater re-applies
#   the overrides on its next poll cycle (~2 minutes).
#
# Tradeoff:
#   Pods show frontend:latest@sha256:<digest> instead of
#   frontend:sha-<commit>. Commit traceability requires looking
#   up the digest in GHCR. The SHA tags still exist in GHCR for
#   audit (CI pushes :sha-<40char>, :sha-<7char>, and :latest
#   on every build).
#
# namePattern matches the ArgoCD Application metadata.name
# exactly: "microservices-demo".
#
# :latest suffix on each imageName is the version constraint
# required by the digest strategy. It tells Image Updater which
# tag to track. Without it, the strategy fails with "cannot use
# digest without a version constraint".
# ---------------------------------------------------------------
apiVersion: argocd-image-updater.argoproj.io/v1alpha1
kind: ImageUpdater
metadata:
  name: microservices-demo-updater
  namespace: argocd
spec:
  applicationRefs:
    - namePattern: "microservices-demo"
      commonUpdateSettings:
        updateStrategy: digest
      images:
        - alias: frontend
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/frontend:latest"
        - alias: cartservice
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/cartservice:latest"
        - alias: checkoutservice
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/checkoutservice:latest"
        - alias: currencyservice
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/currencyservice:latest"
        - alias: emailservice
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/emailservice:latest"
        - alias: paymentservice
          imageName: "ghcr.io/ibtisam-iq/microservices-demo/paymentservice:latest"
        - alias: productcatalogservice
