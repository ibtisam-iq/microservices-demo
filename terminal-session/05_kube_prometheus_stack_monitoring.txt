# Release name: kube-prometheus-stack
# Chart: prometheus-community/kube-prometheus-stack
# Namespace: monitoring
#
# Apply with:
#   helm upgrade -i kube-prometheus-stack prometheus-community/kube-prometheus-stack \
#     --version 86.2.3 -f patch-values.yaml -n monitoring --create-namespace

# ── Alertmanager ────────────────────────────────────────────────────────────
alertmanager:
  alertmanagerSpec:
    # Mount the Kubernetes Secret that holds the Slack incoming webhook URL.
    # Secret is created before installing the chart:
    #   kubectl create secret generic alertmanager-slack-webhook \
    #     --from-literal=slack-webhook-url="<webhook-url>" \
    #     -n monitoring
    secrets:
      - alertmanager-slack-webhook

  config:
    route:
      receiver: 'email-default'     # default for non‑critical alerts

      routes:
        - receiver: 'slack-notification'
          matchers:
            - severity = "critical"

    receivers:
      - name: 'slack-notification'
        slack_configs:
          - api_url_file: /etc/alertmanager/secrets/alertmanager-slack-webhook/slack-webhook-url
            channel: '#alertmanager'
            send_resolved: true

      - name: 'email-default'
        email_configs:
          - to: [EMAIL_ADDRESS]

# HTTPRoute — exposes Grafana through the shared ALB Gateway
# Hostname: grafana.ibtisam.qzz.io
# Backend:  kube-prometheus-stack-grafana Service (port 80)
#           Created by the kube-prometheus-stack Helm chart.
apiVersion: gateway.networking.k8s.io/v1beta1
kind: HTTPRoute
metadata:
  name: grafana-route
  namespace: monitoring
spec:
  hostnames:
    - "grafana.ibtisam.qzz.io"
  parentRefs:
    - group: gateway.networking.k8s.io
      namespace: default
      kind: Gateway
      name: app-alb-gateway
      sectionName: http
    - group: gateway.networking.k8s.io
      namespace: default
      kind: Gateway
      name: app-alb-gateway
      sectionName: https
  rules:
    - backendRefs:
        - name: kube-prometheus-stack-grafana
          port: 80
ubuntu@ip-10-0-101-70:~/platform-engineering-systems/systems/microservices-demo/addons/kube-prometheus$ cat target-grp-grafana.yaml 
# TargetGroupConfiguration — registers Grafana pods as ALB targets
# targetReference matches the Service name created by the chart.
# targetType: ip → ALB route directly to pod IPs (required for ClusterIP Services).
apiVersion: gateway.k8s.aws/v1beta1
kind: TargetGroupConfiguration
metadata:
  name: grafana-tg-config
  namespace: monitoring
spec:
  targetReference:
    name: kube-prometheus-stack-grafana
  defaultConfiguration:
    targetType: ip
